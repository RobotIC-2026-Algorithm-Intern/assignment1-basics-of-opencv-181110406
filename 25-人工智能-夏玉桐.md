import cv2
import numpy as np
#红色
red1_low=np.array([0,100,100])
red1_up=np.array([10,255,255])
red2_low=np.array([170,100,100])
red2_up=np.array([180,255,255])
# 蓝色
blue_low = np.array([100, 120, 50])
blue_up = np.array([130, 255, 255])
# 紫色
purple_low = np.array([130, 80, 50])
purple_up = np.array([160, 255, 255])
#roi
x,y = 180,130
width,height = 250,170
def ballstate(frame):
    #rotate
    frame_rotated=cv2.rotate(frame,cv2.ROTATE_180)
    #roi
    roi=frame_rotated[y:y+height,x:x+width]
    hsv = cv2.cvtColor(roi, cv2.COLOR_BGR2HSV)
    #red
    red_ball1=cv2.inRange(hsv,red1_low,red1_up)
    red_ball2=cv2.inRange(hsv,red2_low,red2_up)
    red_ball=cv2.add(red_ball1,red_ball2)
    red_count=cv2.countNonZero(red_ball)
    #blue
    blue_ball=cv2.inRange(hsv,blue_low,blue_up)
    blue_count=cv2.countNonZero(blue_ball)
    #purple
    purple_ball=cv2.inRange(hsv,purple_low,purple_up)
    purple_count=cv2.countNonZero(purple_ball)
    #color counts
    color_counts = {
        'Red': red_count,
        'Blue': blue_count,
        'Purple': purple_count
    }
    dominant_color = max (color_counts, key=color_counts.get)
    dominant_count =color_counts[dominant_color]
    if dominant_count<1000:
        return "noball"
    else:
        return dominant_color

    # 从output.avi中逐帧读取并将当前帧存为frame
cam = cv2.VideoCapture(r"res\output.avi")
cnt = 0
while True:
    ret, frame = cam.read()
    frame = cv2.rotate(frame, cv2.ROTATE_180)
    if not ret:
        break
    # 获取球的状态
    state = ballstate(frame)        
    # 绘制ROI矩形框
    cv2.rectangle(frame, (x, y), (x + width, y + height), (0, 255, 0), 2)
    # 显示球的状态
    cv2.putText(frame, state, (30, 50), cv2.FONT_HERSHEY_SIMPLEX, 1.2, (255, 255, 255), 3)
    # 展示当前帧
    cv2.imshow('frame', frame)
    cv2.waitKey(25)
    # 在while循环中进行你对frame的处理
cam.release()
